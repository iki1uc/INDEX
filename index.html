function updateStatus(respoCount){
    const box = document.getElementById("statusKaro");
    const txt = document.getElementById("statusText");

    if(respoCount === 410){
        box.style.background = "#0f0"; // Grün
        txt.innerText = "ONLINE – 410 RESPO geladen";
    }
    else if(respoCount > 0){
        box.style.background = "#ff0"; // Gelb
        txt.innerText = "WARNUNG – nur " + respoCount + " RESPO geladen";
    }
    else {
        box.style.background = "#f00"; // Rot
        txt.innerText = "OFFLINE – keine RESPO geladen";
    }
}
export const RESPO_LIST = {

    build(){
        const list = [];

        for(let i=0; i<410; i++){
            list.push({
                id: i,
                tmp: "TMP-" + Math.floor(Math.random()*9999),

                axe: {
                    d: i % 9,
                    hdf: Math.floor(i / 9),
                    o: (i * 7) % 360,
                    u: (i * 13) % 360,
                    gamma: GAM.gamma(),
                    qi: VECTOR.qi(i % 9, Math.floor(i / 9)),
                    iqq: VECTOR.iqq(i % 9, Math.floor(i / 9))
                }
            });
        }

        return list;
    }
};

window.RESPO_LIST = RESPO_LIST;
import { RESPO_LIST } from "./RESPO_LIST.js";

export const RESPO_INDEX = {

    ALL: [],
    ORG: {},
    LOL: {},
    PIPE: [],

    async init(){
        this.ALL = RESPO_LIST.build();

        this.ORG = {
            active: this.ALL.filter(x => x),
            tmpSorted: this.ALL.sort((a,b)=>a.tmp.localeCompare(b.tmp))
        };

        this.LOL = {
            random: this.ALL[Math.floor(Math.random()*410)],
            last10: this.ALL.slice(-10)
        };

        this.PIPE = this.ALL.map(x => x.id);
    }
};

window.RESPO_INDEX = RESPO_INDEX;
